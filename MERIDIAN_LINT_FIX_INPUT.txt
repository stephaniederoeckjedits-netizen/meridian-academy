export default function DynamicResumeMission() {
  const pathname = usePathname()
  const [mounted, setMounted] = useState(false)
  const [mission, setMission] = useState({ label: 'START MISSION', title: 'Dutch Mastery', href: '/courses/english-to-dutch-mastery' })

  const getContextType = useCallback((): ResumeType => {
    if (pathname.startsWith('/courses')) return 'course'
    if (pathname.startsWith('/library/prompts')) return 'spec'
    if (pathname.startsWith('/library/vault')) return 'vault'
    // FIX: Library defaults to Ebook
    if (pathname.startsWith('/library')) return 'ebook' 
    return 'course' // Dashboard defaults to Course
  }, [pathname])

  const resolveContext = useCallback(async () => {
    const type = getContextType()
    const saved = await getResume(type)

    if (saved) {
      const labels = { course: 'RESUME COURSE', ebook: 'RESUME READING', spec: 'RESUME SPEC', vault: 'RESUME VAULT' }
      setMission({ label: labels[type], title: saved.title, href: saved.href })
    } else {
      const fallbacks = {
        course: { label: 'START COURSE', title: 'Dutch Mastery', href: '/courses/english-to-dutch-mastery' },
        ebook: { label: 'EXPLORE LIBRARY', title: 'Browse Ebooks', href: '/library/ebooks' },

        vault: { label: 'EXPLORE VAULT', title: 'Browse Vault', href: '/library/vault' }
      }
      setMission(fallbacks[type])
    }
  }, [getContextType])

  useEffect(() => {
    setMounted(true)
    resolveContext()

    window.addEventListener('meridian-resume-updated', handleUpdate)
    return () => window.removeEventListener('meridian-resume-updated', handleUpdate)

import { useEffect, useRef, useState } from 'react'
import { addPracticeSeconds } from '@/lib/activity'

type SessionType = 'quick' | 'focus' | 'deep'
const SESSIONS: Record<SessionType, { label: string; mins: number; accent: string }> = {
  quick: { label: '15', mins: 15, accent: '#10b981' },
  focus: { label: '25', mins: 25, accent: '#22d3ee' },
  deep:  { label: '50', mins: 50, accent: '#8b5cf6' },
}
const BREAK_MINS = 5

export default function PomodoroTimer({ userEmail }: { userEmail?: string }) {
  const prefix = userEmail ? userEmail + '_' : 'guest_'
  const [type, setType] = useState<SessionType>('focus')
  const [mode, setMode] = useState<'work' | 'break'>('work')
  const [seconds, setSeconds] = useState(SESSIONS.focus.mins * 60)
  const [running, setRunning] = useState(false)
  const [sessions, setSessions] = useState(0)
  const [mounted, setMounted] = useState(false)
  const intervalRef = useRef<ReturnType<typeof setInterval> | null>(null)

  useEffect(() => {
    if (!prefix || prefix === 'guest_') return
    const today = new Date().toDateString()
    const storedDate = localStorage.getItem(prefix + 'm-pomo-date')
    const storedCount = localStorage.getItem(prefix + 'm-pomo-count')
    if (storedDate === today && storedCount) {
      setSessions(parseInt(storedCount))
    } else {
      localStorage.setItem(prefix + 'm-pomo-date', today)
      localStorage.setItem(prefix + 'm-pomo-count', '0')
      setSessions(0)
    }
    setMounted(true)
  }, [prefix])

  useEffect(() => {
    if (!running) return
    intervalRef.current = setInterval(() => {
      if (mode === 'work') { addPracticeSeconds(1, userEmail) }
      setSeconds((s) => {
        if (s <= 1) {
          if (mode === 'work') {
            const today = new Date().toDateString()
            const storedDate = localStorage.getItem(prefix + 'm-pomo-date')
            const storedCount = localStorage.getItem(prefix + 'm-pomo-count')
            const current = storedDate === today && storedCount ? parseInt(storedCount) : 0
            const next = current + 1
            localStorage.setItem(prefix + 'm-pomo-date', today)
            localStorage.setItem(prefix + 'm-pomo-count', String(next))
            setSessions(next)
            setMode('break')
            return BREAK_MINS * 60
          } else {
            setMode('work')
            return SESSIONS[type].mins * 60
          }
        }
        return s - 1
      })
    }, 1000)
    return () => { if (intervalRef.current) clearInterval(intervalRef.current) }
  }, [running, mode, type, prefix])

  function selectType(t: SessionType) {
    setRunning(false); setType(t); setMode('work'); setSeconds(SESSIONS[t].mins * 60)
  }
  function toggle() { setRunning((r) => !r) }
  function reset() { setRunning(false); setMode('work'); setSeconds(SESSIONS[type].mins * 60) }

  if (!mounted) return null

  const total = mode === 'work' ? SESSIONS[type].mins * 60 : BREAK_MINS * 60
  const elapsed = total - seconds
  const pct = (elapsed / total) * 100
  const mins = Math.floor(seconds / 60)
  const secs = seconds % 60
  const isWork = mode === 'work'
  const accent = isWork ? SESSIONS[type].accent : '#f59e0b'
  const R = 40
  const C = 2 * Math.PI * R
  const dash = (pct / 100) * C

  return (
    <div className="brief-box p-4" style={{ '--accent-color': accent } as any}>
      <div className="flex items-center justify-between mb-3">
        <div className="flex items-center gap-2">
          <span className="w-1.5 h-1.5 rounded-full animate-pulse" style={{ background: accent }} />
          <span className="text-[10px] font-mono uppercase tracking-widest font-bold" style={{ color: accent }}>{isWork ? 'FOCUS' : 'BREAK'}</span>
        </div>
        <span className="text-[10px] font-mono text-[#566074]">{sessions} DONE</span>
      </div>
      <div className="grid grid-cols-3 gap-1 mb-4">
        {(Object.keys(SESSIONS) as SessionType[]).map((k) => {
          const s = SESSIONS[k]; const active = k === type
          return (
            <button key={k} onClick={() => selectType(k)} className="text-[9px] font-mono tracking-[0.2em] py-1.5 rounded transition-all" style={{ color: active ? s.accent : '#566074', background: active ? 'rgba(34,211,238,0.06)' : 'transparent', border: '1px solid ' + (active ? s.accent + '55' : 'rgba(34,211,238,0.12)') }}>{s.label} MIN</button>
          )
        })}
      </div>
      <div className="relative flex items-center justify-center mb-4" style={{ height: 110 }}>
        <svg width="110" height="110" className="absolute -rotate-90">
          <circle cx="55" cy="55" r={R} fill="none" stroke="rgba(30,38,51,0.9)" strokeWidth="3" />
          <circle cx="55" cy="55" r={R} fill="none" stroke={accent} strokeWidth="3" strokeLinecap="round" strokeDasharray={C} strokeDashoffset={C - dash} style={{ transition: 'stroke-dashoffset 0.5s linear', filter: 'drop-shadow(0 0 6px ' + accent + 'aa)' }} />
        </svg>
        <div className="text-center relative">
          <div className="text-3xl font-bold tabular-nums leading-none" style={{ color: '#a5f3fc', textShadow: '0 0 20px ' + accent + '66' }}>{String(mins).padStart(2, '0')}:{String(secs).padStart(2, '0')}</div>
          <div className="text-[9px] font-mono tracking-[0.25em] text-[#3a4250] mt-1.5">{isWork ? 'WORK' : 'REST'}</div>

    const w = Array(7).fill(false)
    const jsDay = new Date().getDay()
    const displayIdx = jsDay === 0 ? 6 : jsDay - 1
    const effective = todayDone ? streakVal : Math.max(0, streakVal)
    for (let i = 0; i < 7 && i <= displayIdx; i++) {
      if (i < effective) w[displayIdx - i] = true
    }
    return w
  }

    const storedTotal = localStorage.getItem(prefix + 'm-streak-total')

    setMounted(true)
  }, [])

  function toggle() {
    const next = theme === 'dark' ? 'light' : 'dark'
    setTheme(next)
    localStorage.setItem('m-theme', next)
    document.documentElement.setAttribute('data-theme', next)
  }

  if (!mounted) return null

  if (compact) {
    return (
      <button
        onClick={toggle}
        aria-label="Toggle theme"

import { createClient } from '@/lib/supabase/client'

export default function ReadingProgressTracker({ articleId, userId }: { articleId: string; userId: string }) {
  const [progress, setProgress] = useState(0)
  const [completed, setCompleted] = useState(false)
  const lastSave = useRef<number>(0)
  const supabase = createClient()

  useEffect(() => {
    if (!userId) return

    // Load existing progress
    async function load() {
      const { data } = await supabase
        .from('journal_reading_progress')
        .select('scroll_percentage, completed')
        .eq('user_id', userId)
        .eq('article_id', articleId)
        .maybeSingle()
      if (data) {
        setProgress(data.scroll_percentage || 0)
        setCompleted(data.completed || false)
      }
    }
    load()

    // Track scroll
    function onScroll() {
      const article = document.querySelector('[data-article-body]') as HTMLElement
      if (!article) return
      const rect = article.getBoundingClientRect()
      const total = article.offsetHeight
      const scrolled = Math.max(0, -rect.top + window.innerHeight * 0.3)
      const pct = Math.min(100, Math.round((scrolled / (total - window.innerHeight * 0.7)) * 100))
      
      if (pct > progress) {
        setProgress(pct)
        if (pct >= 90 && !completed) setCompleted(true)
        
        // Debounced save (max once per 3s)
        const now = Date.now()
        if (now - lastSave.current > 3000) {
          lastSave.current = now
          supabase.from('journal_reading_progress').upsert({
            user_id: userId,
            article_id: articleId,
            scroll_percentage: pct,
            completed: pct >= 90,
            updated_at: new Date().toISOString(),
          }, { onConflict: 'user_id,article_id' })
        }
      }
    }

    window.addEventListener('scroll', onScroll, { passive: true })
    onScroll()
    return () => window.removeEventListener('scroll', onScroll)
  }, [articleId, userId, supabase])

  // Top progress bar
  return (
    <div className="fixed top-0 left-0 right-0 h-[3px] z-50 pointer-events-none bg-white/5">
      <div
        className="h-full bg-gradient-to-r from-[#f59e0b] via-[#8b5cf6] to-[#ec4899] transition-[width] duration-150 shadow-[0_0_12px_#f59e0b]"
        style={{ width: progress + '%' }}
      />
    </div>
  )
}

// Tracks daily activity flags used by auto-checked Daily Goals
// Every key is prefixed by user email so accounts never leak into each other

function prefix(userEmail?: string) {
  return userEmail ? userEmail + '_' : 'guest_'
}

function todayKey() {
  return new Date().toISOString().slice(0, 10)
}

export function markCodexOpened(userEmail?: string) {
  try {
    localStorage.setItem(`${prefix(userEmail)}activity_codex_${todayKey()}`, '1')
  } catch (e) {}
}

export function markFlashcardUsed(userEmail?: string) {
  try {
    localStorage.setItem(`${prefix(userEmail)}activity_flashcard_${todayKey()}`, '1')
  } catch (e) {}
}

export function addPracticeSeconds(seconds: number, userEmail?: string) {
  try {
    const key = `${prefix(userEmail)}activity_practice_sec_${todayKey()}`
    const current = parseInt(localStorage.getItem(key) || '0')
    localStorage.setItem(key, String(current + seconds))
  } catch (e) {}
}

export function getPracticeMinutes(userEmail?: string): number {
  try {
    const key = `${prefix(userEmail)}activity_practice_sec_${todayKey()}`
    const seconds = parseInt(localStorage.getItem(key) || '0')
    return Math.floor(seconds / 60)
  } catch (e) { return 0 }
}

export function hasFlag(flag: 'codex' | 'flashcard', userEmail?: string): boolean {
  try {
    return localStorage.getItem(`${prefix(userEmail)}activity_${flag}_${todayKey()}`) === '1'
  } catch (e) { return false }
}

function parseRSS(xml: string) {
  const items: any[] = []
  const rssRegex = /<item>([\s\S]*?)<\/item>/g
  const atomRegex = /<entry>([\s\S]*?)<\/entry>/g
  const regex = rssRegex.test(xml) ? /<item>([\s\S]*?)<\/item>/g : atomRegex
  let match
  while ((match = regex.exec(xml)) !== null) {
    const itemContent = match[1]
    const getTag = (tag: string) => {
      const cdataMatch = itemContent.match(new RegExp(`<${tag}><!\\[CDATA\\[([\\s\\S]*?)\\]\\]></${tag}>`))
      if (cdataMatch) return cdataMatch[1]
      const normalMatch = itemContent.match(new RegExp(`<${tag}[^>]*>([\\s\\S]*?)</${tag}>`))
      return normalMatch ? normalMatch[1].replace(/<[^>]+>/g, '').trim() : ''
    }
    const linkMatch = itemContent.match(/<link[^>]*href="([^"]+)"/)
    const link = linkMatch ? linkMatch[1] : getTag('link')
    let pubDate = getTag('pubDate') || getTag('published') || getTag('updated')
    try {
      const d = new Date(pubDate)
      if (isNaN(d.getTime())) pubDate = new Date().toISOString()
      else pubDate = d.toISOString()
    } catch { pubDate = new Date().toISOString() }

    const title = getTag('title')
    const url = link || getTag('guid')
    if (title && url) {
      items.push({
        title,
        url,
        excerpt: getTag('description').substring(0, 280).replace(/<[^>]+>/g, '').trim() + '...',
        published_at: pubDate,
      })
    }
  }
  return items
}

export async function fetchAndStoreNews() {
  const supabase = createClient(
    process.env.NEXT_PUBLIC_SUPABASE_URL!,
    process.env.SUPABASE_SERVICE_ROLE_KEY!
  )

  // Purge older than 20 days
  const twentyDaysAgo = new Date()
  twentyDaysAgo.setDate(twentyDaysAgo.getDate() - 20)
  await supabase.from('daily_news').delete().lt('published_at', twentyDaysAgo.toISOString())

  // Fetch in parallel
  const results = await Promise.allSettled(
    FEEDS.map(async (feed) => {
      try {
        const response = await fetch(feed.url, {
          headers: { 'User-Agent': 'MeridianAcademy/1.0 (RSS Reader)' },
          signal: AbortSignal.timeout(6000),
        })
        if (!response.ok) return []
        const xml = await response.text()
        const items = parseRSS(xml)
        // FETCH 5 PER FEED (Instead of 2) to guarantee coverage
        return items.slice(0, 5).map(item => ({
          ...item,
          source: feed.source,
          category: feed.category,
        }))
      } catch (e) {
        return []
      }
    })
  )

  const allArticles: any[] = []
  results.forEach(r => {
    if (r.status === 'fulfilled') allArticles.push(...r.value)
  })

  if (allArticles.length === 0) return { count: 0 }

  // Insert in chunks
  let totalInserted = 0
  for (let i = 0; i < allArticles.length; i += 50) {
    const chunk = allArticles.slice(i, i + 50)
    const { error } = await supabase.from('daily_news').upsert(chunk, { onConflict: 'url' })
    if (!error) totalInserted += chunk.length
  }

  return { count: totalInserted, fetched: allArticles.length, sources: FEEDS.length }
}

import { createClient } from '@/lib/supabase/client'

export type ResumeType = 'course' | 'ebook' | 'spec' | 'vault';

async function getPrefix(): Promise<string> {
  const supabase = createClient()
  const { data: { user } } = await supabase.auth.getUser()
  return user ? user.id + '_' : 'guest_'
}

export async function trackResume(
  type: ResumeType, 
  id: string, 
  title: string, 
  href: string, 
  progressData: any = {}
) {
  const supabase = createClient()
  const prefix = await getPrefix()
  
  try {
    const record = { type, id, title, href, timestamp: Date.now(), progressData };
    localStorage.setItem(prefix + 'meridian_resume_' + type, JSON.stringify(record));
    window.dispatchEvent(new Event('meridian-resume-updated'));
  } catch (e) {}

  try {
    const { data: { user } } = await supabase.auth.getUser()
    if (!user) return

    await supabase.from('user_resume_state').upsert({
      user_id: user.id,
      resource_type: type,
      resource_id: id,
      title: title,
      href: href,
      progress_data: progressData,
      updated_at: new Date().toISOString()
    }, { onConflict: 'user_id, resource_type' })
  } catch (err) {
    console.error('[Meridian] DB Sync failed:', err);
  }
}

export async function getResume(type: ResumeType) {
  const prefix = await getPrefix()
  
  try {
    const raw = localStorage.getItem(prefix + 'meridian_resume_' + type);
    if (raw) {
      const parsed = JSON.parse(raw);
      if (parsed.href && parsed.title) return parsed;
    }
  } catch {}

  try {
    const supabase = createClient()
    const { data: { user } } = await supabase.auth.getUser()
    if (!user) return null

    const { data } = await supabase
      .from('user_resume_state')
      .select('*')
      .eq('user_id', user.id)
      .eq('resource_type', type)
      .single()

    if (data) {
      const record = { type, id: data.resource_id, title: data.title, href: data.href, timestamp: Date.now(), progressData: data.progress_data }
      localStorage.setItem(prefix + 'meridian_resume_' + type, JSON.stringify(record))
      return record
    }
  } catch {}
  return null;
}
