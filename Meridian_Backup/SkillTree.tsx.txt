'use client'

import { useEffect, useState } from 'react'
import { createClient } from '@/lib/supabase/client'

type SkillNode = {
  id: string
  name: string
  description: string
  unlocked: boolean
}

const PROGRESSION: SkillNode[] = [
  { id: 'novice', name: 'Digital Novice', description: 'The journey begins.', unlocked: true },
  { id: 'reader', name: 'Avid Reader', description: 'Consumed foundational texts.', unlocked: false },
  { id: 'builder', name: 'Builder', description: 'Applied first principles.', unlocked: false },
  { id: 'systems_thinker', name: 'Systems Thinker', description: 'See invisible feedback loops.', unlocked: false },
  { id: 'language_architect', name: 'Language Architect', description: 'Master communication grids.', unlocked: false },
  { id: 'disciplined_mind', name: 'Disciplined Mind', description: 'Unbreakable focus.', unlocked: false },
  { id: 'systems_architect', name: 'Digital Systems Architect', description: 'Master of complex systems.', unlocked: false },
  { id: 'renaissance', name: 'Digital Renaissance Architect', description: 'The pinnacle of interdisciplinary mastery.', unlocked: false },
]

export default function SkillTree() {
  const [skills, setSkills] = useState<SkillNode[]>(PROGRESSION)
  const [xp, setXp] = useState(0)
  const [level, setLevel] = useState(1)

  useEffect(() => {
    async function loadSkills() {
      const supabase = createClient()
      const { data: { user } } = await supabase.auth.getUser()
      if (!user) return

      const { data: profile } = await supabase.from('profiles').select('xp, level').eq('id', user.id).single()
      if (profile) {
        setXp(profile.xp || 0)
        setLevel(profile.level || 1)
      }

      const { data: userSkills } = await supabase.from('user_skills').select('*').eq('user_id', user.id)

      if (userSkills) {
        const merged = PROGRESSION.map(skill => {
          const dbSkill = userSkills.find(us => us.skill_name === skill.name)
          return dbSkill ? { ...skill, unlocked: dbSkill.unlocked } : skill
        })
        setSkills(merged)
      }
    }
    loadSkills()
  }, [])

  const xpForNextLevel = Math.floor(100 * Math.pow(level, 1.5))
  const progressPercent = Math.min(100, Math.floor((xp / xpForNextLevel) * 100))

  return (
    <div className="w-full flex flex-col gap-5">
      
      {/* Level Progress Bar */}
      <div>
        <div className="flex items-center justify-between mb-2">
          <span className="text-[10px] font-mono text-[#22d3ee] tracking-wider font-bold">LEVEL {level}</span>
          <span className="text-[9px] font-mono text-white">{xp} / {isNaN(xpForNextLevel) ? '---' : xpForNextLevel} XP</span>
        </div>
        <div className="w-full h-2 bg-[#1e2633] rounded-full overflow-hidden border border-[#22d3ee]/20">
          <div className="h-full bg-gradient-to-r from-[#22d3ee] to-[#3b82f6] transition-all duration-1000" style={{ width: `${progressPercent}%` }} />
        </div>
      </div>

      {/* Progression List */}
      <div className="relative border-l border-[#1e2633] ml-2 pl-4 space-y-5 pb-2">
        {skills.map((skill, index) => {
          const isCurrent = !skill.unlocked && (index === 0 || skills[index - 1].unlocked);
          
          return (
            <div key={skill.id} className="relative group">
              {/* Timeline Dot */}
              <div 
                className={`absolute -left-[21px] top-1 w-3 h-3 rounded-full border-2 transition-colors duration-300 ${
                  skill.unlocked 
                    ? 'bg-[#22d3ee] border-[#22d3ee] shadow-[0_0_10px_#22d3ee]' 
                    : isCurrent 
                      ? 'bg-[#0a0f16] border-[#8b5cf6] animate-pulse' 
                      : 'bg-[#0a0f16] border-[#3a4250]'
                }`} 
              />
              
              <div className={`transition-opacity ${skill.unlocked ? 'opacity-100' : 'opacity-60'}`}>
                <div className={`text-[11px] font-bold ${skill.unlocked ? 'text-white' : isCurrent ? 'text-[#8b5cf6]' : 'text-[#566074]'}`}>
                  {skill.name}
                </div>
                <div className="text-[9px] text-[#8892a6] mt-0.5">
                  {skill.description}
                </div>
              </div>
            </div>
          )
        })}
      </div>
    </div>
  )
}
