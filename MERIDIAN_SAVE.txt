import { redirect } from 'next/navigation'
import Link from 'next/link'
import { createClient } from '@/lib/supabase/server'
import AppShell from '@/components/AppShell'
import EnrollButton from './EnrollButton'

const DASHBOARD_HIDDEN_SLUGS = new Set<string>(['english-b1-fluency'])

export default async function DashboardPage() {
  const supabase = await createClient()
  const { data: { user } } = await supabase.auth.getUser()

  if (!user) redirect('/login')

  const { data: profile } = await supabase
    .from('profiles').select('*').eq('id', user.id).single()

  const { data: courses } = await supabase
    .from('courses').select('*').eq('published', true)

  const visibleCourses = (courses ?? []).filter(
    (c) => !DASHBOARD_HIDDEN_SLUGS.has(c.slug)
  )

  const { data: enrollments } = await supabase
    .from('enrollments').select('course_id').eq('user_id', user.id)

  const enrolledIds = new Set(enrollments?.map((e) => e.course_id) || [])

  return (
    <AppShell userEmail={user.email}>
      <div className="max-w-5xl mx-auto px-6 md:px-10 py-12">
        <div className="mb-12">
          <div className="flex items-center gap-2 text-[10px] font-mono tracking-[0.3em] text-[#22d3ee] mb-3">
            <span className="w-1 h-1 rounded-full bg-[#22d3ee] animate-pulse" />
            DASHBOARD
            <span className="w-px h-3 mx-2" style={{ background: 'rgba(34,211,238,0.25)' }} />
            <span className="text-[#3a4250]">OVERVIEW / LIVE</span>
          </div>
          <h1 className="text-3xl md:text-5xl font-bold tracking-[0.06em] text-[#e5e9f0] mb-3 leading-tight">
            Welcome back, {profile?.full_name || 'learner'}
          </h1>
          <div className="flex items-center gap-3">
            <p className="text-sm text-[#8892a6]">Pick a course and continue your journey.</p>
            <span className="flex-1 h-px max-w-md" style={{ background: 'linear-gradient(90deg, rgba(34,211,238,0.35) 0%, transparent 100%)' }} />
          </div>
        </div>

        <div className="flex items-center gap-3 mb-5">
          <span className="text-[10px] font-mono tracking-[0.3em] text-[#566074]">AVAILABLE COURSES</span>
          <span className="w-px h-3" style={{ background: 'rgba(34,211,238,0.2)' }} />
          <span className="text-[10px] font-mono tracking-[0.3em] text-[#3a4250]">
            {String(visibleCourses.length).padStart(2, '0')} LOADED
          </span>
        </div>

        {visibleCourses.length === 0 ? (
          <div className="m-hud-panel p-12 text-center">
            <span className="m-hud-scan" />
            <div className="text-base text-[#e5e9f0] mb-2">No courses available yet.</div>
            <div className="text-sm text-[#8892a6]">New curriculum arrives soon.</div>
          </div>
        ) : (
          <div className="grid md:grid-cols-2 gap-5">

              const enrolled = enrolledIds.has(course.id)
              const trackLabel =
                course.track === 'en_to_nl' ? 'EN → NL'
                : course.track === 'nl_to_en' ? 'NL → EN'
                : course.track?.toUpperCase()

                <div key={course.id} className="m-hud-panel flex flex-col relative transition-all duration-300 hover:translate-x-1 group">
                  <span className="m-hud-scan" />

                    <div className="flex items-center gap-2">

                      <span style={{ color: '#3a4250' }}>/ COURSE</span>
                    </div>

                    <p className="text-[#8892a6] text-sm leading-relaxed">{course.description}</p>
                  </Link>

                  <div className="px-6 md:px-7 pb-6 md:pb-7">
                    {enrolled ? (
                      <Link href={'/courses/' + course.slug}
                        className="inline-flex items-center gap-2 bg-[#22d3ee] text-black text-[11px] font-bold tracking-[0.2em] uppercase rounded-md px-5 py-2.5 hover:bg-cyan-300 transition">
                        Continue Course
                        <span className="text-base leading-none">→</span>

                    ) : (
                      <EnrollButton courseId={course.id} enrolled={false} />

import Link from 'next/link'
import { redirect } from 'next/navigation'
import { createClient } from '@/lib/supabase/server'
import { getCurrentUser } from '@/lib/auth'
import AppShell from '@/components/AppShell'
import ContentArrowNav from '@/components/ContentArrowNav'

type Course = {
  id: string
  slug: string
  title: string
  description: string | null
  track: string
  level: string
}

type Enrollment = {
  course_id: string
  progress: number | null
}

// Courses temporarily hidden from the catalog until their content is ready.
// Remove a slug from here once the course is complete.
// Long-term clean fix: set courses.published = false in Supabase.
const HIDDEN_SLUGS = new Set<string>([
  'english-b1-fluency',
])

export default async function CoursesPage() {
  const user = await getCurrentUser()
  if (!user) redirect('/login')

  const supabase = await createClient()

  const { data: courses } = await supabase
    .from('courses')
    .select('id, slug, title, description, track, level')
    .order('created_at', { ascending: true })

  const visibleCourses = ((courses ?? []) as Course[]).filter(
    c => !HIDDEN_SLUGS.has(c.slug)
  )

  const { data: enrollments } = await supabase
    .from('enrollments')
    .select('course_id, progress')
    .eq('user_id', user.id)

  const enrolledMap = new Map<string, number>()
  ;(enrollments as Enrollment[] | null)?.forEach(e => {
    enrolledMap.set(e.course_id, e.progress ?? 0)
  })

  const list = visibleCourses

  return (
    <AppShell userEmail={user.email ?? undefined}>
      <div className="max-w-4xl mx-auto">

        {/* Masthead */}
        <div className="mb-10 md:mb-12 relative">
          <div className="flex items-center gap-2 text-[10px] font-mono tracking-[0.3em] text-[#22d3ee] mb-3">
            <span className="w-1 h-1 rounded-full bg-[#22d3ee] animate-pulse" />
            COURSES
            <span className="w-px h-3 mx-2" style={{ background: 'rgba(34,211,238,0.25)' }} />
            <span className="text-[#3a4250]">CATALOG / 01</span>
          </div>
          <h1 className="text-2xl sm:text-3xl md:text-5xl font-bold tracking-[0.08em] text-[#e5e9f0] mb-4 leading-tight">
            SELECT YOUR PATH
          </h1>
          <div className="flex items-center gap-3">
            <p className="text-sm text-[#8892a6]">
              {list.length} {list.length === 1 ? 'program' : 'programs'} available
            </p>
            <span className="flex-1 h-px" style={{ background: 'linear-gradient(90deg, rgba(34,211,238,0.35) 0%, transparent 100%)' }} />
          </div>
        </div>

        {list.length === 0 ? (
          <div
            className="rounded-2xl p-12 text-center relative"
            style={{
              border: '1px solid rgba(34,211,238,0.18)',
              background: 'rgba(6,10,16,0.6)',
            }}
          >
            <span
              className="pointer-events-none absolute top-0 left-0 w-3 h-3"
              style={{ borderTop: '1.5px solid #22d3ee', borderLeft: '1.5px solid #22d3ee', opacity: 0.5 }}
            />
            <span
              className="pointer-events-none absolute bottom-0 right-0 w-3 h-3"
              style={{ borderBottom: '1.5px solid #22d3ee', borderRight: '1.5px solid #22d3ee', opacity: 0.5 }}
            />
            <div className="text-lg text-[#e5e9f0] mb-2">No programs yet.</div>
            <div className="text-sm text-[#8892a6]">New curriculum arrives soon.</div>
          </div>
        ) : (
          <ContentArrowNav>
            <div className="space-y-3">
              {list.map((course, i) => {
                const enrolled = enrolledMap.has(course.id)

  title: string
  description: string | null
  order_index: number
  lessons: Lesson[]

export type Saga = {
  id: string
  title: string
  description: string | null
  order_index: number
  missions: Mission[]

type State = 'cleared' | 'active' | 'next' | 'hidden'
type Current = { si: number; mi: number; li: number }

type Props = {
  sagas: Saga[]
  slug: string
  completedIds: string[]
  enrolled: boolean
  defaultOpenIndex: number
}

function computeSagaState(si: number, c: Current): State {
  if (c.si === -1) return 'cleared'
  if (si < c.si) return 'cleared'
  if (si === c.si) return 'active'
  if (si === c.si + 1) return 'next'
  return 'hidden'
}

function computeMissionState(si: number, mi: number, c: Current): State {
  if (si < c.si) return 'cleared'
  if (si > c.si) return 'hidden'
  if (mi < c.mi) return 'cleared'
  if (mi === c.mi) return 'active'
  if (mi === c.mi + 1) return 'next'
  return 'hidden'
}

function computeLessonState(si: number, mi: number, li: number, c: Current): State {
  if (si < c.si) return 'cleared'
  if (si > c.si) return 'hidden'
  if (mi < c.mi) return 'cleared'
  if (mi > c.mi) return 'hidden'
  if (li < c.li) return 'cleared'
  if (li === c.li) return 'active'
  if (li === c.li + 1) return 'next'
  return 'hidden'
}

  sagas, slug, completedIds, enrolled, defaultOpenIndex,

      const missions = sagas[si].missions ?? []
      for (let mi = 0; mi < missions.length; mi++) {
        const lessons = missions[mi].lessons ?? []
        for (let li = 0; li < lessons.length; li++) {
          if (!completedSet.has(lessons[li].id)) return { si, mi, li }
        }
      }
    }
    return { si: -1, mi: -1, li: -1 }
  }, [sagas, completedSet])

  const [openSagaIdx, setOpenSagaIdx] = useState(defaultOpenIndex)

  useEffect(() => {
    if (current.si >= 0) {
      setOpenSagaIdx(current.si)
      const s = sagas[current.si]
      if (s?.missions?.[current.mi]) {
        setOpenMissionKey(s.missions[current.mi].id)
      }
    }
  }, [current.si, current.mi, sagas])

  return (
    <div className="space-y-3 md:space-y-4">
      {sagas.map((saga, si) => {

            current={current}
            openMissionKey={openMissionKey}
            setOpenMissionKey={setOpenMissionKey}
          />
        )
      })}

}

import { redirect, notFound } from 'next/navigation'
import { createClient } from '@/lib/supabase/server'
import { getCurrentUser } from '@/lib/auth'

import QuestPlayer from './QuestPlayer'

export default async function LessonPage({
  params,
}: {
  params: Promise<{ slug: string; lessonId: string }>
}) {
  const { slug, lessonId } = await params
  const user = await getCurrentUser()
  if (!user) redirect('/login')

  const supabase = await createClient()

  const { data: course } = await supabase
    .from('courses')
    .select('id, slug, title')
    .eq('slug', slug)
    .maybeSingle()
  if (!course) notFound()

  const { data: lesson } = await supabase
    .from('lessons')
    .select(
      'id, title, order_index, video_url, content, is_final_exam, pass_threshold, mission_id, mission:missions(id, order_index, module_id, module:modules(id, order_index))'
    )
    .eq('id', lessonId)
    .maybeSingle()
  if (!lesson) notFound()

  const mission = (lesson as any).mission as
    | { id: string; order_index: number; module_id: string; module: { id: string; order_index: number } | null }
    | null

  let nextHref: string | null = null
  let nextTitle: string | null = null

  if (mission) {
    const { data: siblings } = await supabase
      .from('lessons')
      .select('id, title')
      .eq('mission_id', mission.id)
      .gt('order_index', lesson.order_index)
      .order('order_index', { ascending: true })
      .limit(1)

    if (siblings && siblings.length > 0) {
      nextHref = `/courses/${slug}/lessons/${siblings[0].id}`
      nextTitle = siblings[0].title
    } else {
      const { data: nextMissions } = await supabase
        .from('missions')
        .select('id')
        .eq('module_id', mission.module_id)
        .gt('order_index', mission.order_index)
        .order('order_index', { ascending: true })
        .limit(1)

      if (nextMissions && nextMissions.length > 0) {
        const { data: firstLesson } = await supabase
          .from('lessons')
          .select('id, title')
          .eq('mission_id', nextMissions[0].id)
          .order('order_index', { ascending: true })
          .limit(1)

        if (firstLesson && firstLesson.length > 0) {
          nextHref = `/courses/${slug}/lessons/${firstLesson[0].id}`
          nextTitle = firstLesson[0].title
        }
      } else {
        const currentSagaOrder = mission.module?.order_index ?? 0
        const { data: nextSagas } = await supabase
          .from('modules')
          .select('id')
          .eq('course_id', course.id)
          .gt('order_index', currentSagaOrder)
          .order('order_index', { ascending: true })
          .limit(1)

        if (nextSagas && nextSagas.length > 0) {
          const { data: nextMissions2 } = await supabase
            .from('missions')
            .select('id')
            .eq('module_id', nextSagas[0].id)
            .order('order_index', { ascending: true })
            .limit(1)

          if (nextMissions2 && nextMissions2.length > 0) {
            const { data: firstLesson2 } = await supabase
              .from('lessons')
              .select('id, title')
              .eq('mission_id', nextMissions2[0].id)
              .order('order_index', { ascending: true })
              .limit(1)

            if (firstLesson2 && firstLesson2.length > 0) {
              nextHref = `/courses/${slug}/lessons/${firstLesson2[0].id}`
              nextTitle = firstLesson2[0].title
            }
          }
        }
      }
    }
  }

  const { data: progress } = await supabase
    .from('lesson_progress')
    .select('completed')
    .eq('user_id', user.id)
    .eq('lesson_id', lessonId)
    .maybeSingle()

  return (
    <AppShell userEmail={user.email ?? undefined}>
      <div className="max-w-3xl mx-auto pb-12 md:pb-20">
        <QuestPlayer
          lessonId={lesson.id}
          lessonTitle={lesson.title}
          lessonIndex={lesson.order_index}

          content={lesson.content}
          isFinalExam={lesson.is_final_exam ?? false}
          passThreshold={lesson.pass_threshold ?? 80}
          initialCompleted={progress?.completed ?? false}
          slug={slug}
          nextHref={nextHref}
          nextTitle={nextTitle}
        />
      </div>
    </AppShell>
  )
}

function repair(raw: string): string {
  let s = raw
  s = s.replace(/\|\s+\|/g, '|\n|')
  s = s.replace(/([^\n])\s+(\*\s+\*[A-Za-z])/g, '$1\n$2')
  s = s.replace(/([^\n])\s+(#{1,3}\s)/g, '$1\n$2')
  s = s.replace(/([^\n])\s+(>\s)/g, '$1\n$2')
  s = s.replace(/\n{3,}/g, '\n\n')
  return s
}

function inline(text: string, keyBase: string): ReactNode[] {
  const out: ReactNode[] = []
  const regex = /(\*\*[^*]+\*\*|\*[^*\n]+\*|`[^`]+`)/g
  text.split(regex).forEach((part, idx) => {
    if (!part) return
    const k = `${keyBase}-${idx}`
    if (part.startsWith('**') && part.endsWith('**') && part.length > 4) {
      const word = part.slice(2, -2)
      const isKeyword = /^[A-Z][A-Z\s\-]{0,22}$/.test(word) && word.length <= 24
      if (isKeyword) {
        out.push(
          <span key={k} className="font-bold tracking-[0.04em]"
            style={{ color: '#a5f3fc', textShadow: '0 0 8px rgba(34,211,238,0.25)' }}>
            {word}
          </span>
        )
      } else {
        out.push(
          <strong key={k} style={{ color: '#ffffff', fontWeight: 700 }}>{word}</strong>
        )
      }
    }
    else if (part.startsWith('*') && part.endsWith('*') && part.length > 2)
      out.push(<em key={k} style={{ color: '#a5f3fc', fontStyle: 'italic' }}>{part.slice(1, -1)}</em>)
    else if (part.startsWith('`') && part.endsWith('`'))
      out.push(
        <code key={k} className="px-1.5 py-0.5 rounded text-[0.9em] font-mono"
          style={{ background: 'var(--m-surface-2)', color: '#a5f3fc' }}>
          {part.slice(1, -1)}
        </code>
      )
    else out.push(<span key={k}>{part}</span>)
  })
  return out
}

function tryVerticalSequence(lines: string[], start: number): { pairs: { value: string; label: string }[]; end: number } | null {
  const pairs: { value: string; label: string }[] = []
  let i = start
  while (i + 2 < lines.length) {
    const v = lines[i].trim()
    const arrow = lines[i + 1].trim()
    const label = lines[i + 2].trim()
    if (!v || !label) break
    if (!ARROW_RE.test(arrow)) break
    if (v.startsWith('#') || v.startsWith('>') || /^[-*] /.test(v)) break
    if (isTableRow(v)) break
    if (v.length > 40 || label.length > 30) break
    pairs.push({ value: v, label })
    i += 3
    while (i < lines.length && lines[i].trim() === '') i++
  }
  if (pairs.length < 2) return null
  return { pairs, end: i }
}

function tryEqualsGroup(lines: string[], start: number): { items: { left: string; right: string }[]; end: number } | null {
  const items: { left: string; right: string }[] = []
  let i = start
  while (i < lines.length) {
    const raw = lines[i].trim()
    if (raw === '') { i++; continue }
    const line = stripBulletPrefix(raw)
    const m = line.match(EQ_SPLIT_RE)
    if (!m) break
    const left = m[1].trim()
    const right = m[2].trim()
    if (!left || !right) break
    if (left.length > 40 || right.length > 40) break
    items.push({ left, right })
    i++
    while (i < lines.length && lines[i].trim() === '') i++
  }
  if (items.length < 2) return null
  return { items, end: i }
}

function tryEmDashPairs(lines: string[], start: number): { items: { left: string; right: string }[]; end: number } | null {
  const items: { left: string; right: string }[] = []
  let i = start
  while (i < lines.length) {
    const raw = lines[i].trim()
    if (raw === '') { i++; continue }

    const right = m[2].trim()
    if (!left || !right) break
    if (left.length > 60 || right.length > 60) break
    if (left.includes('→') || right.includes('→')) break
    items.push({ left, right })
    i++
    while (i < lines.length && lines[i].trim() === '') i++
  }
  if (items.length < 2) return null
  return { items, end: i }
}

function tryNumberedList(lines: string[], start: number): { items: string[]; end: number } | null {
  const items: string[] = []
  let i = start

}

export default function CodexContent({ content }: { content: string }) {
  const cleaned = repair(content)
  const lines = cleaned.split('\n')
  const blocks: ReactNode[] = []
  let i = 0
  let key = 0

  while (i < lines.length) {
    const line = lines[i]
    const trimmed = line.trim()

    if (trimmed === '') { i++; continue }

    if (isTableRow(line) && i + 1 < lines.length && isTableSeparator(lines[i + 1])) {
      const headers = splitRow(line)
      i += 2
      const rows: string[][] = []
      while (i < lines.length && isTableRow(lines[i])) {
        const cells = splitRow(lines[i])
        if (cells.some(c => c.length > 0)) rows.push(cells)
        i++
      }
      blocks.push(<DataTable key={key++} headers={headers} rows={rows} k={String(key)} />)
      continue
    }

    if (trimmed.startsWith('### ')) {
      blocks.push(
        <h3 key={key++} className="text-base md:text-lg font-bold tracking-wide mt-10 md:mt-12 mb-3"
          style={{ color: '#ffffff' }}>
          {inline(trimmed.slice(4), `h3-${key}`)}
        </h3>
      )
      i++
      continue
    }
    if (trimmed.startsWith('## ')) {
      blocks.push(<SectionHeading key={key++} text={trimmed.slice(3)} k={String(key)} />)
      i++
      continue
    }
    if (trimmed.startsWith('# ')) {
      blocks.push(
        <h1 key={key++}
          className="text-2xl sm:text-3xl md:text-4xl font-bold tracking-tight mt-4 md:mt-6 mb-5 md:mb-6"
          style={{ color: '#ffffff' }}>
          {inline(trimmed.slice(2), `h1-${key}`)}
        </h1>
      )
      i++
      continue
    }

    if (trimmed === '---') {
      blocks.push(
        <div key={key++} className="my-10 md:my-12 flex items-center gap-3" aria-hidden>
          <span className="flex-1 h-px" style={{ background: 'var(--m-border)' }} />
          <span className="w-1 h-1 rounded-full" style={{ background: 'rgba(34,211,238,0.6)' }} />
          <span className="flex-1 h-px" style={{ background: 'var(--m-border)' }} />
        </div>
      )
      i++
      continue
    }

    if (trimmed.startsWith('> ')) {
      const quoted: string[] = []
      while (i < lines.length && lines[i].trim().startsWith('> ')) {
        quoted.push(lines[i].trim().slice(2))
        i++
      }

      const single = quoted.length === 1 ? quoted[0] : ''
      const hasStructure = /[→←↓↑=]/.test(single)
      const isCompact = quoted.length === 1 && single.length > 0 && single.length <= 55 && !hasStructure

      if (quoted.length === 1 && single.includes('→') && single.length <= 90) {
        const chainParts = single.split('→').map(s => s.trim()).filter(Boolean)
        if (chainParts.length >= 2 && chainParts.length <= 6 && chainParts.every(p => p.length <= 30)) {
          blocks.push(<ChainBlock key={key++} parts={chainParts} k={`ch-${key}`} />)
          continue
        }
      }

      if (isCompact) {
        blocks.push(<CompactLabel key={key++} text={single} k={`cl-${key}`} />)
        continue
      }

      blocks.push(
        <blockquote key={key++}
          className="my-6 md:my-8 relative rounded-xl px-5 md:px-6 py-4 md:py-5 overflow-hidden"
          style={{
            background: 'linear-gradient(135deg, rgba(34,211,238,0.07) 0%, rgba(59,130,246,0.02) 100%)',
            border: '1px solid rgba(34,211,238,0.22)',
          }}>
          <span className="absolute left-0 top-0 bottom-0 w-[2px]"
            style={{ background: 'rgba(34,211,238,0.55)' }} />
          {quoted.map((q, qi) => (
            <p key={qi} className="text-[15px] md:text-base leading-relaxed" style={{ color: '#ffffff' }}>
              {inline(q, `q-${key}-${qi}`)}
            </p>
          ))}
        </blockquote>
      )
      continue
    }

    const vs = tryVerticalSequence(lines, i)
    if (vs) {
      blocks.push(<VerticalSequence key={key++} pairs={vs.pairs} k={`vs-${key}`} />)
      i = vs.end
      continue
    }

    const eq = tryEqualsGroup(lines, i)
    if (eq) {
      blocks.push(<Pairs key={key++} items={eq.items} k={`eq-${key}`} />)
      i = eq.end
      continue
    }

    const ed = tryEmDashPairs(lines, i)
    if (ed) {
      blocks.push(<Pairs key={key++} items={ed.items} k={`ed-${key}`} />)
      i = ed.end
      continue
    }

    const nl = tryNumberedList(lines, i)
    if (nl) {
      blocks.push(
        <ol key={key++} className="my-5 md:my-6 space-y-2.5 md:space-y-3">
          {nl.items.map((it, ii) => (
            <li key={ii} className="flex items-start gap-3 md:gap-4">
              <span className="text-[11px] font-mono tracking-widest shrink-0 pt-[3px] w-6 tabular-nums"
                style={{ color: '#a5f3fc', opacity: 0.9 }}>
                {String(ii + 1).padStart(2, '0')}
              </span>
              <span className="text-[15px] md:text-base leading-relaxed flex-1" style={{ color: '#ffffff' }}>
                {inline(it, `nl-${key}-${ii}`)}
              </span>
            </li>
          ))}
        </ol>
      )
      i = nl.end
      continue
    }

    if (/^[-*] /.test(trimmed)) {
      const items: string[] = []
      while (i < lines.length && /^[-*] /.test(lines[i].trim()) && !isTableRow(lines[i])) {
        items.push(lines[i].trim().slice(2))
        i++
      }
      blocks.push(
        <div key={key++} className="my-5 md:my-6 relative pl-6 md:pl-8 space-y-2"
          style={{ borderLeft: '1px solid var(--m-border-bright)' }}>
          {items.map((it, ii) => (
            <div key={ii} className="relative">
              <span className="absolute -left-[27px] md:-left-[35px] top-[18px] w-1.5 h-1.5 rounded-full"
                style={{ background: 'rgba(34,211,238,0.75)' }} />
              <BulletItem text={it} k={`li-${key}-${ii}`} />
            </div>
          ))}
        </div>
      )
      continue
    }

    if (ARROW_RE.test(trimmed)) {
      blocks.push(
        <div key={key++} className="flex my-3 pl-4">
          <span className="text-xl leading-none" style={{ color: 'rgba(34,211,238,0.5)' }}>↓</span>
        </div>
      )
      i++
      continue
    }

    const circM = trimmed.match(CIRCLED_RE)
    if (circM) {
      const num = CIRCLE_MAP[circM[1]] ?? circM[1]
      const label = circM[2].replace(/\*\*/g, '').replace(/\*/g, '').trim()
      if (label.length > 0 && label.length <= 40) {
        blocks.push(<StepHeading key={key++} num={num} text={label} k={String(key)} />)
        i++
        continue
      }
    }

    const chainParts = trimmed.split('→').map(s => s.trim())
    const isCompact2 = chainParts.length === 2 &&
      chainParts[0].length > 0 && chainParts[0].length <= 25 &&
      chainParts[1].length > 0 && chainParts[1].length <= 25
    const isBigChain = chainParts.length >= 3 &&
      chainParts.length <= 5 &&
      chainParts.every(p => p.length > 0 && p.length < 40)
    if (isBigChain || isCompact2) {
      blocks.push(<ChainBlock key={key++} parts={chainParts} k={`ch-${key}`} compact={isCompact2} />)
      i++
      continue
    }

    const para: string[] = [trimmed]
    i++
    while (
      i < lines.length &&
      lines[i].trim() !== '' &&
      !lines[i].trim().startsWith('#') &&
      !lines[i].trim().startsWith('>') &&
      !/^[-*] /.test(lines[i].trim()) &&
      lines[i].trim() !== '---' &&
      !isTableRow(lines[i]) &&
      !ARROW_RE.test(lines[i].trim()) &&
      !CIRCLED_RE.test(lines[i].trim())
    ) {
      para.push(lines[i].trim())
      i++
    }
    blocks.push(
      <p key={key++} className="my-4 md:my-5 text-[15px] md:text-base leading-relaxed" style={{ color: '#ffffff' }}>
        {inline(para.join(' '), `p-${key}`)}
      </p>
    )
  }

  return <div className="max-w-[68ch]">{blocks}</div>
}

  isFinalExam: boolean
  passThreshold: number
  initialCompleted: boolean
  slug: string
  nextHref: string | null
  nextTitle: string | null
}) {
  const router = useRouter()
  const supabase = createClient()

  const [phase, setPhase] = useState<Phase>(initialCompleted ? 'mission' : 'briefing')
  const [cameFromResult, setCameFromResult] = useState(false)
  const [questions, setQuestions] = useState<Question[] | null>(null)
  const [answers, setAnswers] = useState<Record<string, number>>({})
  const [examIndex, setExamIndex] = useState(0)
  const [submitting, setSubmitting] = useState(false)
  const [result, setResult] = useState<QuizResult | null>(null)
  const [error, setError] = useState<string | null>(null)
  const [completed, setCompleted] = useState(initialCompleted)
  const [showDiscipline, setShowDiscipline] = useState(false)

  useEffect(() => {
    if (questions !== null) return
    async function load() {
      const { data, error } = await supabase.rpc('get_quiz_questions', {
        p_lesson_id: lessonId,
      })
      if (error) { setError(error.message); setQuestions([]); return }
      setQuestions((data as Question[]) || [])
    }
    load()
  }, [lessonId, questions, supabase])

  useEffect(() => {
    requestAnimationFrame(() => {
      requestAnimationFrame(() => scrollMainToTop(false))
    })
  }, [phase])

  function scrollTop() { scrollMainToTop(true) }
  function startMission() { setPhase('mission'); scrollTop() }
  function finishMission() {
    setPhase('exam'); setAnswers({}); setResult(null); setError(null)
    setExamIndex(0); setCameFromResult(false); scrollTop()
  }
  function retry() {
    setPhase('exam'); setAnswers({}); setResult(null); setError(null)
    setExamIndex(0); setCameFromResult(false); scrollTop()
  }
  function backToCodexFromResult() {
    setPhase('mission')
    setCameFromResult(true)
    setError(null)
    requestAnimationFrame(() => requestAnimationFrame(() => scrollMainToTop(false)))
  }
  function backToResult() {
    setPhase('result')
    setCameFromResult(false)
    requestAnimationFrame(() => requestAnimationFrame(() => scrollMainToTop(false)))
  }

  function goNext(href: string) {
    triggerNav()
    setTimeout(() => router.push(href), 300)
  }

  async function submit() {
    setSubmitting(true); setError(null)
    const payload = questions && questions.length > 0 ? answers : {}
    const { data, error: rpcError } = await supabase.rpc('submit_quiz', {
      p_lesson_id: lessonId, p_answers: payload,
    })
    if (rpcError) { setError(rpcError.message); setSubmitting(false); return }

    let parsed: unknown = data
    if (typeof parsed === 'string') { try { parsed = JSON.parse(parsed) } catch { parsed = null } }
    let r: QuizResult | null = null
    if (Array.isArray(parsed)) r = (parsed[0] as QuizResult) ?? null
    else if (parsed && typeof parsed === 'object' && 'passed' in (parsed as object)) r = parsed as QuizResult

    if (!r || typeof r.passed !== 'boolean') {
      setError('Exam result was malformed. Please retry.'); setSubmitting(false); return
    }

    if (r.passed) {
      playPassSound()
      setCompleted(true)
      setShowDiscipline(true)
    } else {
      playFailSound()
    }

    setResult(r)
    setPhase('result')
    setSubmitting(false)
    requestAnimationFrame(() => requestAnimationFrame(() => scrollMainToTop(false)))
  }

  const allAnswered =
    questions !== null && questions.length > 0 &&
    questions.every(q => answers[q.id] !== undefined)

  const unansweredCount =
    questions && questions.length > 0
      ? questions.length - Object.keys(answers).length
      : 0

  /* BRIEFING */
  if (phase === 'briefing') {
    return (
      <div className="m-hud-panel">
        <span className="m-hud-scan" />
        <div className="m-hud-header">
          <div className="flex items-center gap-2">
            <span className="w-1 h-1 rounded-full bg-[#22d3ee] animate-pulse" />
            MISSION BRIEF
            <span style={{ color: '#3a4250' }}>/ {isFinalExam ? 'FINAL' : 'SQ ' + String(lessonIndex).padStart(2, '0')}</span>
          </div>
          <div className="flex items-center gap-2">
            <span className="w-1.5 h-1.5 rounded-full bg-[#22d3ee] animate-pulse" />
            STANDBY
          </div>
        </div>
        <div className="p-6 sm:p-8 md:p-10">
        <div className="m-label m-label-cyan mb-3">
          {isFinalExam ? 'Final Exam' : 'Side Quest ' + String(lessonIndex).padStart(2, '0')}
        </div>
        <h2 className="text-2xl sm:text-3xl md:text-4xl font-bold tracking-tight mb-8">{lessonTitle}</h2>
        <div className="border-t border-[var(--m-border)] mb-6" />
        <div className="grid grid-cols-1 sm:grid-cols-3 gap-5 mb-10">
          <div>
            <div className="text-[10px] font-mono tracking-widest text-[var(--m-text-dim)] mb-1">BRIEFING</div>
            <div className="text-xl sm:text-2xl font-bold text-[var(--m-text)]">{videoUrl ? 'READY' : 'TEXT ONLY'}</div>
          </div>
          <div>
            <div className="text-[10px] font-mono tracking-widest text-[var(--m-text-dim)] mb-1">EXAM</div>
            <div className="text-xl sm:text-2xl font-bold text-[var(--m-text)]">
              {questions === null ? '...' : questions.length === 0 ? 'NONE' : questions.length + ' Q'}
            </div>
          </div>
          <div>
            <div className="text-[10px] font-mono tracking-widest text-[var(--m-text-dim)] mb-1">PASS GATE</div>
            <div className="text-xl sm:text-2xl font-bold text-[var(--m-cyan)]">{passThreshold}%</div>
          </div>
        </div>
        <div className="m-label mb-4">Objectives</div>
        <ul className="space-y-2 mb-10 text-sm">
          <li className="flex gap-3 text-[var(--m-text)]"><span className="text-[var(--m-cyan)] shrink-0">▸</span><span>Study the Quest Briefing</span></li>
          <li className="flex gap-3 text-[var(--m-text)]"><span className="text-[var(--m-cyan)] shrink-0">▸</span><span>Read the Quest Codex</span></li>
          <li className="flex gap-3 text-[var(--m-text)]"><span className="text-[var(--m-cyan)] shrink-0">▸</span><span>Score {passThreshold}% or higher on the exam</span></li>
          <li className="flex gap-3 text-[var(--m-text)]"><span className="text-[var(--m-cyan)] shrink-0">▸</span><span>Unlock the next quest — +50 XP</span></li>
        </ul>
        <button onClick={startMission} className="m-pulse w-full md:w-auto bg-[var(--m-cyan)] text-black font-bold tracking-widest text-sm rounded-lg px-10 py-4 hover:bg-cyan-300 transition">
          ▶ START MISSION
        </button>
        </div>
      </div>
    )
  }

  /* MISSION */
  if (phase === 'mission') {
    const showReview = cameFromResult && result?.review && questions

    if (showReview) {
      return (
        <div className="space-y-6">
          <div className="m-panel relative overflow-hidden">
            <span className="pointer-events-none absolute top-0 left-0 w-4 h-4 z-10" style={{ borderTop: '1.5px solid #22d3ee', borderLeft: '1.5px solid #22d3ee' }} />
            <span className="pointer-events-none absolute top-0 right-0 w-4 h-4 z-10" style={{ borderTop: '1.5px solid #22d3ee', borderRight: '1.5px solid #22d3ee' }} />
            <span className="pointer-events-none absolute bottom-0 left-0 w-4 h-4 z-10" style={{ borderBottom: '1.5px solid #22d3ee', borderLeft: '1.5px solid #22d3ee' }} />
            <span className="pointer-events-none absolute bottom-0 right-0 w-4 h-4 z-10" style={{ borderBottom: '1.5px solid #22d3ee', borderRight: '1.5px solid #22d3ee' }} />
            <span className="pointer-events-none absolute left-0 right-0 top-0 h-px z-10" style={{ background: 'linear-gradient(90deg, transparent 0%, #22d3ee 50%, transparent 100%)', opacity: 0.75 }} />
            <div className="px-5 md:px-7 py-4 md:py-5 flex items-center justify-between gap-4 flex-wrap">
              <div className="min-w-0">
                <div className="flex items-center gap-2 mb-2">
                  <span className="w-1 h-1 rounded-full bg-[#22d3ee] animate-pulse" />
                  <span className="text-[10px] font-mono tracking-[0.3em]" style={{ color: '#22d3ee', textShadow: '0 0 10px rgba(34,211,238,0.55)' }}>
                    {isFinalExam ? 'FINAL EXAM' : 'SIDE QUEST ' + String(lessonIndex).padStart(2, '0')}
                  </span>
                </div>
                <div className="text-base md:text-xl font-bold tracking-tight text-[var(--m-text)] truncate">
                  {lessonTitle}
                </div>
              </div>
              <button onClick={backToResult}
                className="text-[10px] font-mono tracking-[0.25em] text-[#22d3ee] hover:text-white transition border border-[rgba(34,211,238,0.4)] rounded-lg px-4 py-2 shrink-0">
                ← BACK TO RESULT
              </button>
            </div>
          </div>

          <div className="text-center py-2">
            <div className="inline-flex items-center gap-2 text-[10px] font-mono tracking-[0.4em] text-[var(--m-cyan)]">
              <span className="w-1 h-1 rounded-full bg-[var(--m-cyan)] animate-pulse" />
              EXAM REVIEW · LEARN FROM YOUR ANSWERS
              <span className="w-1 h-1 rounded-full bg-[var(--m-cyan)] animate-pulse" />
            </div>
          </div>

          {questions.map((q, qi) => {
            const rv = result.review!.find(r => r.id === q.id)
            if (!rv) return null
            const userIdx = rv.user_index
            const correctIdx = rv.correct_index
            const isCorrect = userIdx === correctIdx
            return (
              <div key={q.id} className="m-panel relative p-4 sm:p-6"
                style={{ borderColor: isCorrect ? 'rgba(16,185,129,0.4)' : 'rgba(239,68,68,0.4)' }}>
                <span className="pointer-events-none absolute top-0 left-0 w-3 h-3"
                  style={{ borderTop: '1.5px solid ' + (isCorrect ? '#10b981' : '#ef4444'), borderLeft: '1.5px solid ' + (isCorrect ? '#10b981' : '#ef4444') }} />
                <span className="pointer-events-none absolute top-0 right-0 w-3 h-3"
                  style={{ borderTop: '1.5px solid ' + (isCorrect ? '#10b981' : '#ef4444'), borderRight: '1.5px solid ' + (isCorrect ? '#10b981' : '#ef4444') }} />
                <span className="pointer-events-none absolute bottom-0 left-0 w-3 h-3"
                  style={{ borderBottom: '1.5px solid ' + (isCorrect ? '#10b981' : '#ef4444'), borderLeft: '1.5px solid ' + (isCorrect ? '#10b981' : '#ef4444') }} />
                <span className="pointer-events-none absolute bottom-0 right-0 w-3 h-3"
                  style={{ borderBottom: '1.5px solid ' + (isCorrect ? '#10b981' : '#ef4444'), borderRight: '1.5px solid ' + (isCorrect ? '#10b981' : '#ef4444') }} />

                <div className="flex items-start justify-between gap-4 mb-4">
                  <div className="flex items-center gap-3">
                    <span className="text-[10px] font-mono tracking-widest text-[var(--m-text-dim)]">
                      Q{String(qi + 1).padStart(2, '0')}
                    </span>
                    <span className="text-lg" style={{ color: isCorrect ? '#10b981' : '#ef4444' }}>
                      {isCorrect ? '✓' : '✗'}
                    </span>
                  </div>
                  <span className="text-[10px] font-mono tracking-widest"
                    style={{ color: isCorrect ? '#10b981' : '#ef4444' }}>
                    {isCorrect ? 'CORRECT' : 'INCORRECT'}
                  </span>
                </div>

                <div className="text-sm md:text-base font-semibold mb-4 text-[var(--m-text)]">
                  {q.question}
                </div>

                <div className="space-y-1.5">
                  {q.options.map((opt, oi) => {
                    const isUser = oi === userIdx
                    const isAnswer = oi === correctIdx
                    let border = 'var(--m-border)'
                    let bg = 'transparent'
                    let color = 'var(--m-text-muted)'
                    let marker = ''
                    if (isAnswer) { border = 'rgba(16,185,129,0.6)'; bg = 'rgba(16,185,129,0.08)'; color = '#10b981'; marker = '✓ correct' }
                    else if (isUser && !isCorrect) { border = 'rgba(239,68,68,0.6)'; bg = 'rgba(239,68,68,0.08)'; color = '#ef4444'; marker = '✗ your answer' }
                    return (
                      <div key={oi} className="flex items-center gap-3 text-xs md:text-sm rounded-lg px-3 py-2.5"
                        style={{ border: '1px solid ' + border, background: bg, color }}>
                        <span className="font-mono shrink-0 w-5">{String.fromCharCode(65 + oi)}</span>
                        <span className="flex-1">{opt}</span>
                        {marker && <span className="text-[10px] font-mono tracking-wider shrink-0">{marker}</span>}
                      </div>
                    )
                  })}
                </div>

                {!isCorrect && (
                  <div className="mt-4 pt-4 border-t text-xs md:text-sm" style={{ borderColor: 'rgba(34,211,238,0.20)' }}>
                    <div className="flex items-start gap-2">
                      <span className="text-[var(--m-cyan)] font-mono tracking-widest shrink-0">💡 TIP</span>
                      <span className="text-[var(--m-text-muted)]">
                        Correct answer is <span style={{ color: '#10b981' }}>{String.fromCharCode(65 + correctIdx)}</span>. Re-read the question — the pattern matters.
                      </span>
                    </div>
                  </div>
                )}
              </div>
            )
          })}

          <button onClick={backToResult}
            className="w-full border border-[var(--m-cyan)]/40 text-[var(--m-cyan)] font-bold tracking-widest text-sm rounded-lg px-8 py-4 hover:bg-[var(--m-cyan-soft)] transition">
            ← BACK TO RESULT
          </button>
        </div>
      )
    }

    return (
      <div className="space-y-8 md:space-y-10">
        <div className="m-panel relative overflow-hidden">
          <span className="pointer-events-none absolute top-0 left-0 w-4 h-4 z-10" style={{ borderTop: '1.5px solid #22d3ee', borderLeft: '1.5px solid #22d3ee' }} />
          <span className="pointer-events-none absolute top-0 right-0 w-4 h-4 z-10" style={{ borderTop: '1.5px solid #22d3ee', borderRight: '1.5px solid #22d3ee' }} />
          <span className="pointer-events-none absolute bottom-0 left-0 w-4 h-4 z-10" style={{ borderBottom: '1.5px solid #22d3ee', borderLeft: '1.5px solid #22d3ee' }} />
          <span className="pointer-events-none absolute bottom-0 right-0 w-4 h-4 z-10" style={{ borderBottom: '1.5px solid #22d3ee', borderRight: '1.5px solid #22d3ee' }} />
          <span className="pointer-events-none absolute left-0 right-0 top-0 h-px z-10" style={{ background: 'linear-gradient(90deg, transparent 0%, #22d3ee 50%, transparent 100%)', opacity: 0.75 }} />

          <div className="px-5 md:px-7 py-4 md:py-5 flex items-center justify-between gap-4 flex-wrap">
            <div className="min-w-0">
              <div className="flex items-center gap-2 mb-2">
                <span className="w-1 h-1 rounded-full bg-[#22d3ee] animate-pulse" />
                <span className="text-[10px] font-mono tracking-[0.3em]" style={{ color: '#22d3ee', textShadow: '0 0 10px rgba(34,211,238,0.55)' }}>
                  {isFinalExam ? 'FINAL EXAM' : 'SIDE QUEST ' + String(lessonIndex).padStart(2, '0')}
                </span>
              </div>
              <div className="text-base md:text-xl font-bold tracking-tight text-[var(--m-text)] truncate">
                {lessonTitle}
              </div>
            </div>
            <div className="flex items-center gap-2 text-[10px] font-mono tracking-[0.25em] text-[var(--m-cyan)] shrink-0">
              <span className="w-1.5 h-1.5 rounded-full bg-[var(--m-cyan)] animate-pulse" />
              IN PROGRESS
            </div>
          </div>
        </div>

        {videoUrl && (
          <div>
            <div className="flex items-center gap-3 mb-4 md:mb-5">
              <span className="inline-block w-1 h-5 rounded-full shrink-0" style={{ background: '#22d3ee', boxShadow: '0 0 12px rgba(34,211,238,0.7)' }} />
              <span className="text-xs md:text-sm font-mono tracking-[0.3em] uppercase" style={{ color: '#22d3ee' }}>
                QUEST BRIEFING
              </span>
              <span className="flex-1 h-px" style={{ background: 'linear-gradient(90deg, rgba(34,211,238,0.4) 0%, transparent 100%)' }} />
            </div>
            <div className="w-full aspect-video rounded-xl overflow-hidden bg-black border border-[var(--m-border)]">
              {videoUrl.includes('youtube') || videoUrl.includes('youtu.be') ? (
                <iframe src={'https://www.youtube.com/embed/' + extractYouTubeId(videoUrl)} title={lessonTitle}
                  allow="accelerometer; autoplay; clipboard-write; encrypted-media; gyroscope; picture-in-picture"
                  allowFullScreen className="w-full h-full" />
              ) : (<video src={videoUrl} controls className="w-full h-full" />)}
            </div>
          </div>
        )}

        {content && (
          <div>
            <div className="flex items-center gap-3 mb-4 md:mb-5">
              <span className="inline-block w-1 h-5 rounded-full shrink-0" style={{ background: '#22d3ee', boxShadow: '0 0 12px rgba(34,211,238,0.7)' }} />
              <span className="text-xs md:text-sm font-mono tracking-[0.3em] uppercase" style={{ color: '#22d3ee' }}>
                {isFinalExam ? 'EXAM INSTRUCTIONS' : 'QUEST CODEX'}
              </span>
              <span className="flex-1 h-px" style={{ background: 'linear-gradient(90deg, rgba(34,211,238,0.4) 0%, transparent 100%)' }} />
            </div>
            <div className="m-hud-panel">
              <span className="m-hud-scan" />
              <div className="m-hud-header">
                <div className="flex items-center gap-2">
                  <span className="w-1 h-1 rounded-full bg-[#22d3ee] animate-pulse" />
                  CODEX
                  <span style={{ color: '#3a4250' }}>/ {isFinalExam ? 'EXAM BRIEF' : 'LESSON BODY'}</span>
                </div>
                <span style={{ color: '#3a4250', opacity: 0.7 }}>MERIDIAN / v1</span>
              </div>
              <div className="p-5 sm:p-6 md:p-8"><CodexContent content={content} /></div>
            </div>
          </div>
        )}

        {completed ? (
          <div className="m-panel border-green-800/50 bg-green-950/20 p-5 md:p-8 flex flex-col md:flex-row md:items-center md:justify-between gap-5">
            <div className="flex items-center gap-3 text-green-400 font-semibold">
              <span className="text-2xl">✓</span>
              <span className="tracking-wider">MISSION COMPLETE</span>
            </div>
            <div className="flex flex-col sm:flex-row items-stretch sm:items-center gap-3">
              <button onClick={finishMission}
                className="text-sm text-green-400/70 hover:text-green-400 underline text-center px-4 py-2">
                Retake exam
              </button>
              {nextHref && (
                <button onClick={() => goNext(nextHref)}
                  className="bg-[var(--m-cyan)] text-black font-bold tracking-widest text-xs rounded-lg px-6 py-3 hover:bg-cyan-300 transition text-center">
                  NEXT: {(nextTitle?.toUpperCase() || 'CONTINUE')} →
                </button>
              )}
            </div>
          </div>
        ) : (
          <button onClick={finishMission}
            className="m-hud-corners w-full bg-[var(--m-cyan)] text-black font-bold tracking-widest text-sm rounded-lg px-10 py-4 hover:bg-cyan-300 transition">
            ✓ FINISH MISSION — TAKE EXAM →
          </button>
        )}
      </div>
    )
  }

  /* EXAM */
  if (phase === 'exam') {
    if (questions === null) {
      return (
        <div className="m-panel m-hud-corners p-12 text-center text-[var(--m-text-dim)] font-mono text-sm">
          LOADING EXAM...
        </div>
      )
    }
    if (questions.length === 0) {
      return (
        <div className="space-y-6">
          <div className="m-panel m-hud-corners p-6 md:p-8">
            <div className="m-label mb-2">{isFinalExam ? 'Final Exam' : 'Practice Exam'}</div>
            <p className="text-[var(--m-text-muted)] mb-6">No questions yet. The instructor is preparing this exam.</p>
            <button onClick={submit} disabled={submitting}
              className="bg-white text-black font-bold tracking-widest text-sm rounded-lg px-8 py-3 hover:bg-neutral-200">
              {submitting ? 'COMPLETING...' : '✓ COMPLETE MISSION'}
            </button>
          </div>
        </div>
      )
    }

    const answeredCount = Object.keys(answers).length
    const progressPct = Math.round((answeredCount / questions.length) * 100)
    const q = questions[examIndex]
    const isLast = examIndex === questions.length - 1
    const isFirst = examIndex === 0
    const selectedForQ = answers[q.id]

    return (
      <div className="space-y-3 md:space-y-6">
        <div className="m-panel relative overflow-hidden">
          <span className="pointer-events-none absolute top-0 left-0 w-3 h-3 z-10" style={{ borderTop: '1.5px solid #22d3ee', borderLeft: '1.5px solid #22d3ee' }} />
          <span className="pointer-events-none absolute top-0 right-0 w-3 h-3 z-10" style={{ borderTop: '1.5px solid #22d3ee', borderRight: '1.5px solid #22d3ee' }} />
          <span className="pointer-events-none absolute bottom-0 left-0 w-3 h-3 z-10" style={{ borderBottom: '1.5px solid #22d3ee', borderLeft: '1.5px solid #22d3ee' }} />
          <span className="pointer-events-none absolute bottom-0 right-0 w-3 h-3 z-10" style={{ borderBottom: '1.5px solid #22d3ee', borderRight: '1.5px solid #22d3ee' }} />

          <div className="px-4 md:px-6 py-3 md:py-4 border-b" style={{ borderColor: 'rgba(34,211,238,0.15)' }}>
            <div className="flex items-center justify-between mb-2 md:mb-3">
              <div className="flex items-center gap-2 text-[10px] font-mono tracking-[0.25em] text-[var(--m-cyan)]">
                <span className="w-1 h-1 rounded-full bg-[var(--m-cyan)] animate-pulse" />
                {isFinalExam ? 'FINAL EXAM' : 'PRACTICE EXAM'}
              </div>
              <div className="text-[10px] font-mono tracking-[0.2em] text-[var(--m-text-dim)]">
                Q{String(examIndex + 1).padStart(2, '0')}/{String(questions.length).padStart(2, '0')}
              </div>
            </div>
            <div className="h-[3px] rounded-full overflow-hidden" style={{ background: '#1e2633' }}>
              <div className="h-full transition-all duration-300"
                style={{ width: progressPct + '%', background: 'linear-gradient(90deg, #22d3ee, #8b5cf6)', boxShadow: '0 0 10px rgba(34,211,238,0.6)' }} />
            </div>
            <div className="flex items-center justify-between mt-2 text-[9px] md:text-[10px] font-mono tracking-widest text-[var(--m-text-dim)]">
              <span>{answeredCount}/{questions.length} ANSWERED</span>
              <span>PASS {passThreshold}%</span>
            </div>
          </div>

          <div className="px-4 md:px-8 py-5 md:py-8">
            <div className="text-base md:text-xl font-bold mb-5 md:mb-6 leading-relaxed text-[var(--m-text)]">
              {q.question}
            </div>
            <div className="space-y-2">
              {q.options.map((opt, idx) => {
                const selected = selectedForQ === idx
                return (
                  <button key={idx} type="button"
                    onClick={() => setAnswers(a => ({ ...a, [q.id]: idx }))}
                    className={'w-full text-left text-sm md:text-base rounded-lg px-3 md:px-4 py-3 md:py-3.5 border transition-all flex items-center gap-3 ' + (selected ? 'border-[var(--m-cyan)] bg-[rgba(34,211,238,0.10)] text-[var(--m-text)]' : 'border-[var(--m-border)] hover:border-[var(--m-border-bright)] text-[var(--m-text-muted)] hover:bg-white/[0.02]')}

function extractYouTubeId(url: string): string | null {
  const match = url.match(
    /(?:youtube\.com\/(?:watch\?v=|embed\/|shorts\/)|youtu\.be\/)([a-zA-Z0-9_-]{11})/
  )
  return match ? match[1] : null
}

'use client'
import { useState } from 'react'
import { createClient } from '@/lib/supabase/client'

type Props = {
  email: string
  username: string
  fullName: string
  role: string
  joinedAt: string
}

export default function ProfileClient({
  email,
  username,
  fullName,
  role,
  joinedAt,
}: Props) {
  const supabase = createClient()

  const [name, setName] = useState(fullName)
  const [savingName, setSavingName] = useState(false)
  const [nameMsg, setNameMsg] = useState<{ type: 'ok' | 'err'; text: string } | null>(null)

  const [password, setPassword] = useState('')
  const [savingPw, setSavingPw] = useState(false)
  const [pwMsg, setPwMsg] = useState<{ type: 'ok' | 'err'; text: string } | null>(null)

  async function saveName() {
    setSavingName(true)
    setNameMsg(null)
    const { error } = await supabase
      .from('profiles')
      .update({ full_name: name })
      .eq('email', email)

    if (error) {
      setNameMsg({ type: 'err', text: error.message })
    } else {
      setNameMsg({ type: 'ok', text: 'Name updated.' })
    }
    setSavingName(false)
  }

  async function updatePassword() {
    if (password.length < 6) {
      setPwMsg({ type: 'err', text: 'Password must be at least 6 characters.' })
      return
    }
    setSavingPw(true)
    setPwMsg(null)
    const { error } = await supabase.auth.updateUser({ password })

    if (error) {
      setPwMsg({ type: 'err', text: error.message })
    } else {
      setPwMsg({ type: 'ok', text: 'Password updated.' })
      setPassword('')
    }
    setSavingPw(false)
  }

  return (
    <div className="space-y-4">
      {/* IDENTITY */}
      <div
        className="relative rounded-2xl p-6 md:p-8 overflow-hidden"
        style={{
          border: '1px solid rgba(34,211,238,0.28)',
          background: 'rgba(6,10,16,0.72)',
          backdropFilter: 'blur(20px)',
          boxShadow: '0 0 60px rgba(34,211,238,0.10)',
        }}
      >
        <span className="pointer-events-none absolute top-0 left-0 w-3 h-3"
          style={{ borderTop: '1.5px solid #22d3ee', borderLeft: '1.5px solid #22d3ee' }} />
        <span className="pointer-events-none absolute top-0 right-0 w-3 h-3"
          style={{ borderTop: '1.5px solid #22d3ee', borderRight: '1.5px solid #22d3ee' }} />
        <span className="pointer-events-none absolute bottom-0 left-0 w-3 h-3"
          style={{ borderBottom: '1.5px solid #22d3ee', borderLeft: '1.5px solid #22d3ee' }} />
        <span className="pointer-events-none absolute bottom-0 right-0 w-3 h-3"
          style={{ borderBottom: '1.5px solid #22d3ee', borderRight: '1.5px solid #22d3ee' }} />

        <div className="flex items-center gap-2 text-[10px] font-mono tracking-[0.3em] text-[#22d3ee] mb-5">
          <span className="w-1 h-1 rounded-full bg-[#22d3ee] animate-pulse" />
          IDENTITY
        </div>

        <div className="space-y-4">
          <InfoRow label="EMAIL" value={email} mono />
          <InfoRow label="USERNAME" value={username || '—'} mono />
          <InfoRow label="ROLE" value={role.toUpperCase()} />
          {joinedAt && (
            <InfoRow
              label="JOINED"
              value={new Date(joinedAt).toLocaleDateString('en-GB', {
                day: 'numeric',
                month: 'long',
                year: 'numeric',
              })}
            />
          )}
        </div>
      </div>

      {/* DISPLAY NAME */}
      <div
        className="relative rounded-2xl p-6 md:p-8 overflow-hidden"
        style={{
          border: '1px solid rgba(34,211,238,0.15)',
          background: 'rgba(6,10,16,0.6)',
          backdropFilter: 'blur(20px)',
        }}
      >
        <span className="pointer-events-none absolute top-0 left-0 w-2.5 h-2.5"
          style={{ borderTop: '1px solid #22d3ee', borderLeft: '1px solid #22d3ee' }} />
        <span className="pointer-events-none absolute top-0 right-0 w-2.5 h-2.5"

        <div className="flex items-center gap-2 text-[10px] font-mono tracking-[0.3em] text-[#22d3ee] mb-5">
          <span className="w-1 h-1 rounded-full bg-[#22d3ee] animate-pulse" />
          DISPLAY NAME
        </div>

        <label className="o-label block mb-2 text-[10px]">FULL NAME</label>
        <input
          value={name}
          onChange={e => setName(e.target.value)}
          placeholder="Your name"
          className="o-input"
        />

        <div className="flex items-center gap-3 mt-4">
          <button
            onClick={saveName}
            disabled={savingName || name === fullName}
            className="bg-[var(--m-cyan)] text-black font-bold tracking-widest text-xs rounded-lg px-6 py-3 hover:bg-cyan-300 transition disabled:opacity-40 disabled:cursor-not-allowed"
          >

          {nameMsg && (

              style={{ color: nameMsg.type === 'ok' ? '#10b981' : '#ef4444' }}>
              {nameMsg.type === 'ok' ? '✓ ' : '✗ '}{nameMsg.text}
            </span>
          )}
        </div>
      </div>

      {/* PASSWORD */}
      <div
        className="relative rounded-2xl p-6 md:p-8 overflow-hidden"
        style={{
          border: '1px solid rgba(34,211,238,0.15)',
          background: 'rgba(6,10,16,0.6)',
          backdropFilter: 'blur(20px)',

          SECURITY

        <input
          type="password"

'use client'
import { useEffect, useRef, useState } from 'react'
import { getActiveZone, setActiveZone, onZoneChange } from '@/lib/navZone'
import { unlockAudio, playSelectSound, playConfirmSound } from '@/lib/sounds'

export default function ContentArrowNav({
  children,
}: {
  children: React.ReactNode
}) {
  const ref = useRef<HTMLDivElement>(null)
  const [activeIndex, setActiveIndex] = useState(0)
  const [zoneActive, setZoneActive] = useState(getActiveZone() === 'content')

  useEffect(() => {
    return onZoneChange(z => {
      setZoneActive(z === 'content')
      if (z !== 'content') setActiveIndex(-1)
    })
  }, [])

  useEffect(() => {
    if (!getActiveZone()) setActiveZone('content')
    setZoneActive(getActiveZone() === 'content')
    setActiveIndex(0)
  }, [])

  useEffect(() => {
    if (!zoneActive) return

    function onKey(e: KeyboardEvent) {
      const el = ref.current
      if (!el) return
      const items = el.querySelectorAll<HTMLElement>('[data-nav-item]')
      if (items.length === 0) return

      const active = document.activeElement as HTMLElement | null
      if (active) {
        const tag = active.tagName.toLowerCase()
        if (tag === 'input' || tag === 'textarea' || tag === 'select' || active.isContentEditable) {
          return
        }
      }

      if (e.key === 'ArrowDown' || e.key === 'ArrowRight') {
        e.preventDefault()
        unlockAudio()
        playSelectSound()
        setActiveIndex(i => (i + 1 >= items.length ? 0 : i + 1))
      } else if (e.key === 'ArrowUp' || e.key === 'ArrowLeft') {
        e.preventDefault()
        unlockAudio()
        playSelectSound()
        setActiveIndex(i => (i - 1 < 0 ? items.length - 1 : i - 1))
      } else if (e.key === 'Enter') {
        const idx = activeIndex < 0 ? 0 : activeIndex
        const target = items[idx]
        if (target) {
          e.preventDefault()
          unlockAudio()
          playConfirmSound()
          target.click()
        }
      }
    }
    document.addEventListener('keydown', onKey, true)
    return () => document.removeEventListener('keydown', onKey, true)
  }, [zoneActive, activeIndex])

  useEffect(() => {
    const el = ref.current
    if (!el) return
    const items = el.querySelectorAll<HTMLElement>('[data-nav-item]')
    items.forEach((it, i) => {
      if (zoneActive && i === activeIndex) {
        it.setAttribute('data-nav-active', 'true')
      } else {
        it.removeAttribute('data-nav-active')
      }
    })
  }, [activeIndex, zoneActive])

  return (
    <div ref={ref} data-nav-zone="content">
      {children}
    </div>
  )

const EVENT_NAME = 'meridian:nav-start'

export function triggerNav() {
  if (typeof window === 'undefined') return
  const fn = (window as unknown as { __meridianNav?: () => void }).__meridianNav
  if (typeof fn === 'function') {
    fn()
  } else {
    window.dispatchEvent(new Event(EVENT_NAME))
  }
}

export default function NavLoader() {
  const pathname = usePathname()
  const [visible, setVisible] = useState(false)
  const [fading, setFading] = useState(false)
  const [progress, setProgress] = useState(0)
  const [quote, setQuote] = useState<DisciplineQuote | null>(null)
  const [dots, setDots] = useState('')

  const prevPathRef = useRef(pathname)
  const startedAtRef = useRef(0)
  const progressRef = useRef<ReturnType<typeof setInterval> | null>(null)
  const fallbackRef = useRef<ReturnType<typeof setTimeout> | null>(null)

  useEffect(() => {
    function clearProgress() {
      if (progressRef.current) {
        clearInterval(progressRef.current)
        progressRef.current = null
      }
    }
    function clearFallback() {
      if (fallbackRef.current) {
        clearTimeout(fallbackRef.current)
        fallbackRef.current = null
      }
    }

    function start() {
      clearProgress()
      clearFallback()
      setQuote(pickQuote())
      setProgress(0)
      setFading(false)
      setVisible(true)
      startedAtRef.current = Date.now()

      unlockAudio()
      playTransitionSound(1400)

      progressRef.current = setInterval(() => {
        const elapsed = Date.now() - startedAtRef.current
        const p = Math.min(88, Math.round((elapsed / 2200) * 88))
        setProgress(p)
        if (p >= 88 && progressRef.current) {
          clearInterval(progressRef.current)
          progressRef.current = null
        }
      }, 30)

      fallbackRef.current = setTimeout(() => {
        setProgress(100)
        setFading(true)
        setTimeout(() => {
          setVisible(false)
          setFading(false)
        }, 320)
      }, 8000)
    }

    ;(window as unknown as { __meridianNav?: () => void }).__meridianNav = start
    window.addEventListener(EVENT_NAME, start)

    return () => {
      window.removeEventListener(EVENT_NAME, start)
      delete (window as unknown as { __meridianNav?: () => void }).__meridianNav
      clearProgress()
      clearFallback()
    }
  }, [])

  useEffect(() => {
    if (prevPathRef.current === pathname) return
    prevPathRef.current = pathname
    if (!visible) return

    const paintTimer = setTimeout(() => {
      setProgress(100)
      setFading(true)
      setTimeout(() => {
        setVisible(false)
        setFading(false)
      }, 400)
    }, 250)

    return () => clearTimeout(paintTimer)
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [pathname])

  useEffect(() => {
    if (!visible) return
    const id = setInterval(() => setDots(d => (d.length >= 3 ? '' : d + '.')), 320)
    return () => clearInterval(id)
  }, [visible])

  if (!visible || !quote) return null

  return (
    <div
      className="fixed inset-0 flex flex-col items-center justify-center px-8 pointer-events-none"

            const storedDate = localStorage.getItem('m-pomo-date')
            const storedCount = localStorage.getItem('m-pomo-count')

            setSessions(next)
            setMode('break')
            return BREAK_MINS * 60
          } else {

            return SESSIONS[type].mins * 60
          }

type SoundsStore = {
  ctx: AudioContext | null
  unlocked: boolean
}

function getStore(): SoundsStore {
  if (typeof window === 'undefined') {
    return { ctx: null, unlocked: false }
  }
  const w = window as unknown as { __meridianSounds?: SoundsStore }
  if (!w.__meridianSounds) {
    w.__meridianSounds = { ctx: null, unlocked: false }
  }
  return w.__meridianSounds
}

function getCtx(): AudioContext | null {
  if (typeof window === 'undefined') return null
  const store = getStore()
  if (!store.ctx) {
    const w = window as unknown as {
      AudioContext?: typeof AudioContext
      webkitAudioContext?: typeof AudioContext
    }
    const Ctor = w.AudioContext || w.webkitAudioContext
    if (!Ctor) return null
    store.ctx = new Ctor()
  }
  if (store.ctx.state === 'suspended') {
    store.ctx.resume().catch(() => {})
  }
  return store.ctx
}

export function unlockAudio() {
  const store = getStore()
  if (store.unlocked) return
  const ctx = getCtx()
  if (!ctx) return
  if (ctx.state === 'suspended') {
    ctx.resume().then(() => { store.unlocked = true }).catch(() => {})
  } else {
    store.unlocked = true
  }
}

export function isAudioUnlocked() {
  const store = getStore()
  return store.unlocked && store.ctx !== null && store.ctx.state === 'running'
}

export type NavZone = 'sidebar' | 'content'

type ZoneStore = {
  active: NavZone
  listeners: Set<(z: NavZone) => void>
}

function getStore(): ZoneStore {
  if (typeof window === 'undefined') {
    return { active: 'content', listeners: new Set() }
  }
  const w = window as unknown as { __meridianZone?: ZoneStore }
  if (!w.__meridianZone) {
    w.__meridianZone = { active: 'content', listeners: new Set() }
  }
  return w.__meridianZone
}

export function getActiveZone(): NavZone {
  return getStore().active
}

export function setActiveZone(zone: NavZone) {
  const store = getStore()
  if (store.active === zone) return
  store.active = zone
  store.listeners.forEach((l) => l(zone))
}

export function onZoneChange(cb: (z: NavZone) => void) {
  const store = getStore()
  store.listeners.add(cb)
  return () => {
    store.listeners.delete(cb)
  }
}

export function installZoneTracker() {
  if (typeof window === 'undefined') return () => {}
  function onClick(e: MouseEvent) {
    const target = e.target as HTMLElement | null
    if (!target) return
    const el = target.closest('[data-nav-zone]') as HTMLElement | null
    if (!el) return
    const zone = el.getAttribute('data-nav-zone')
    if (zone === 'sidebar' || zone === 'content') {
      setActiveZone(zone)
    }
  }
  document.addEventListener('click', onClick, true)
  return () => document.removeEventListener('click', onClick, true)
}

import { createServerClient } from '@supabase/ssr'
import { cookies } from 'next/headers'

export async function createClient() {
  const cookieStore = await cookies()

  return createServerClient(
    process.env.NEXT_PUBLIC_SUPABASE_URL!,
    process.env.NEXT_PUBLIC_SUPABASE_ANON_KEY!,
    {
      cookies: {
        getAll() {
          return cookieStore.getAll()
        },
        setAll(cookiesToSet) {
          try {
            cookiesToSet.forEach(({ name, value, options }) =>
              cookieStore.set(name, value, options)
            )
          } catch {}
        },
      },
    }
  )
}
